# [12] Nori 인덱스 만들기 — comment는 ko(Nori), comment.std는 standard
PUT reviews_nori
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0,
    "analysis": {
      "tokenizer": {
        "ko_tokenizer": { "type": "nori_tokenizer", "decompound_mode": "mixed", "user_dictionary_rules": ["가성비"] }
      },
      "analyzer": {
        "ko": { "type": "custom", "tokenizer": "ko_tokenizer", "filter": ["nori_part_of_speech", "nori_readingform", "lowercase"] }
      }
    }
  },
  "mappings": {
    "properties": {
      "review_id": { "type": "keyword" },
      "order_id":  { "type": "keyword" },
      "product":   { "type": "keyword" },
      "region":    { "type": "keyword" },
      "rating":    { "type": "integer" },
      "comment":   { "type": "text", "analyzer": "ko",
                     "fields": { "std": { "type": "text", "analyzer": "standard" } } }
    }
  }
}
